fn test_quest_dialogue_smith_rock() {

fn test_quest_dialogue_smith_rock() {

}

## 4. QuestManager API 参考

**需要的方法** (quests.rs):

```rust
impl QuestManager {
    /// Check if quest is active
    pub fn is_active(&self, id: QuestId) -> bool {
        if let Some(quest) = self.quests.get(id as usize) {
            quest.state.is_active()
        } else {
            false
        }
    }

    /// Set quest state (for testing)
    pub fn set_quest_state(&mut self, id: QuestId, state: QuestState) {
        if let Some(quest) = self.quests.get_mut(id as usize) {
            quest.state = state;
        }
    }

    /// Get quest by ID
    pub fn get_quest(&self, id: QuestId) -> Option<&Quest> {
        self.quests.get(id as usize)
    }
}
```
