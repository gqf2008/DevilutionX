### 关键算法

#### 1. 命中判定算法

```cpp
// PlrHitMonst() lines 529-555
bool PlrHitMonst(Player &player, Monster &monster, bool adjacentDamage) {
    int hper = 0;

    // 不可击中检查
    if (!monster.isPossibleToHit())
        return false;

    // 邻接伤害惩罚
    if (adjacentDamage) {
        if (player.level > 20)
            hper -= 30;
        else
            hper -= (35 - player.level) * 2;
    }

    // 投骰
    int hit = GenerateRnd(100);
    if (monster.mode == MonsterMode::Petrified) {
        hit = 0;  // 石化自动命中
    }

    // 计算命中率
    hper += player.GetMeleePiercingToHit() - player.CalculateArmorPierce(monster.armorClass, true);
    hper = clamp(hper, 5, 95);

    // 判定
    if (hit >= hper) {
        return false;  // 未命中
    }

    // ... 伤害计算
}
```

**关键点**:
- `GenerateRnd(100)` 生成 0-99 的随机数
- 命中率强制限制在 5-95% (永远有 miss 和 hit 的可能)
- 石化怪物跳过随机判定直接命中
- 邻接伤害 (顺劈) 有显著命中率惩罚

#### 2. 伤害计算算法

```cpp
// PlrHitMonst() lines 556-655
// 基础伤害
const int mind = player._pIMinDam;
const int maxd = player._pIMaxDam;
int dam = RandomIntBetween(mind, maxd);

// 百分比加成
dam += dam * player._pIBonusDam / 100;

// 固定加成
dam += player._pIBonusDamMod;
int dam2 = dam << 6;  // 保存用于 Peril

// 属性加成
dam += player._pDamageMod;

// 暴击 (Warrior/Barbarian)
if (HasCriticalStrike(player._pClass)) {
    if (GenerateRnd(100) < player.level) {
        dam *= 2;
    }
}

// 武器类型加成
ItemType weapon = GetWeaponType(player);
switch (monster.monsterClass) {
case MonsterClass::Undead:
    if (weapon == Sword) dam -= dam / 2;  // -50%
    if (weapon == Mace) dam += dam / 2;   // +50%
    break;
case MonsterClass::Animal:
    if (weapon == Mace) dam -= dam / 2;
    if (weapon == Sword) dam += dam / 2;
    break;
case MonsterClass::Demon:
    if (HasTripleDemonDamage(player))
        dam *= 3;
    break;
}

// 特殊词缀
if (HasDevastation && GenerateRnd(100) < 5) {
    dam *= 3;
}

// 转换为固定点数
dam <<= 6;

// Jesters 随机倍率
if (HasJesters) {
    int r = GenerateRnd(201);
    if (r >= 100)
        r = 100 + (r - 100) * 5;
    dam = dam * r / 100;
}

// 邻接伤害惩罚
if (adjacentDamage)
    dam >>= 2;  // 75% 减少

// Peril 反噬
if (HasPeril) {
    int selfDam = dam2 + (player._pIGetHit << 6);
    ApplyPlrDamage(player, selfDam);
    dam *= 2;
}

// 应用伤害
ApplyMonsterDamage(monster, dam);
```

**关键点**:
- 伤害计算顺序严格按照 C++ 原版
- 固定点数 (`<< 6`) 用于精确计算 (1 单位 = 1/64 HP)
- 邻接伤害最后减少 75% (`>> 2` = 除以 4)
- Jesters 可以产生 0-500% 的伤害波动

```rust
/// 执行攻击逻辑 (碰撞检测)
/// C++ Reference: player.cpp:779-840
pub fn do_attack(player: &mut Player) -> bool {
    let mut did_hit = false;
    let attack_pos = player.position + player.direction.offset();

    // 检查主目标
    if let Some(monster) = get_monster_at(attack_pos) {
        did_hit = CombatSystem::player_hit_monster(
            &player.combat_stats(),
            &monster.combat_stats(),
            false,  // not adjacent
            is_hellfire(),
            &mut rng,
        ).is_hit();
    }

    // 顺劈攻击 (邻接目标)
    if player.has_cleave() {
        for adjacent_pos in get_adjacent_positions(attack_pos) {
            if let Some(monster) = get_monster_at(adjacent_pos) {
                CombatSystem::player_hit_monster(
                    &player.combat_stats(),
                    &monster.combat_stats(),
                    true,  // adjacent damage
                    is_hellfire(),
                    &mut rng,
                );
            }
        }
    }

    did_hit
}
```

```rust
/// 玩家攻击玩家 (PvP)
/// C++ Reference: player.cpp:707-781
pub fn player_hit_player(
    attacker: &Player,
    target: &mut Player,
    rng: &mut impl Rng,
) -> bool {
    // 无敌检查
    if target.is_invincible() || target.has_etherealize() {
        return false;
    }

    // 命中判定
    let hit = rng.gen_range(0..100);
    let mut hper = attacker.get_melee_to_hit() - target.get_armor();
    hper = hper.clamp(5, 95);

    if hit >= hper {
        return false;
    }

    // 格挡判定
    let blk = rng.gen_range(0..100);
    let mut blkper = target.get_block_chance() - (attacker.level * 2);
    blkper = blkper.clamp(0, 100);

    if target.is_standing_or_attacking() && target.is_blocking() && blk < blkper {
        start_player_block(target, direction_to(target.position, attacker.position));
        return true;
    }

    // 伤害计算 (简化,完整实现在 Day 2)
    let dam = calculate_pvp_damage(attacker, rng);
    apply_player_damage(target, dam);

    true
}
```

```rust
// Step 3: Flat bonus
dam += player.bonus_dam_mod;

// Save for Peril calculation
let dam2 = dam << 6;

// Step 4: Stat modifier
dam += player.damage_mod;

// Step 5: Critical strike
if player.has_critical_strike() && rng.gen_range(0..100) < player.level {
    dam *= 2;
}

// Step 6: Weapon type bonus
let weapon_type = player.get_weapon_type();
match monster.monster_class {
    MonsterClass::Undead => {
        if weapon_type == ItemType::Sword {
            dam -= dam / 2;
        } else if weapon_type == ItemType::Mace {
            dam += dam / 2;
        }
    }
    MonsterClass::Animal => {
        if weapon_type == ItemType::Mace {
            dam -= dam / 2;
        } else if weapon_type == ItemType::Sword {
            dam += dam / 2;
        }
    }
    MonsterClass::Demon => {
        if player.has_triple_demon_damage() {
            dam *= 3;
        }
    }
    _ => {}
}

// Step 7: Special effects - Devastation (5% chance for 3x damage)
if player.has_devastation() && rng.gen_range(0..100) < 5 {
    dam *= 3;
}

// Step 8: Convert to fixed-point
dam <<= 6;

// Step 9: Jesters modifier (0-500% variance)
if player.has_jesters() {
    let mut r = rng.gen_range(0..201);
    if r >= 100 {
        r = 100 + (r - 100) * 5;
    }
    dam = dam * r / 100;
}

// Step 10: Adjacent damage penalty
if adjacent_damage {
    dam >>= 2;
}

// Step 11: Peril (2x damage but self-damage)
if player.has_peril() {
    let self_dam = dam2 + (player.get_hit << 6);
    if self_dam >= 0 {
        ApplyPlrDamage(player, self_dam);
    }
    dam *= 2;
}
```

```rust
pub fn apply_player_damage(
    player: &mut Player,
    damage: i32,
    damage_type: DamageType,
    armor_pierce: bool,
) {
    let mut dam = damage;

    // 护甲减免 (如果非穿透)
    if !armor_pierce {
        let armor = player.get_armor();
        dam -= dam * armor / 100;
        dam = dam.max(0);
    }

    // 应用伤害
    player.hit_points -= dam >> 6;
    player.hp_base -= dam >> 6;

    if player.hit_points < 0 {
        player.hit_points = 0;
        player.hp_base = 0;
        // 死亡处理
    }

    // 刷新 UI
    redraw_component(PanelDrawComponent::Health);
}
```
