### 3.2 核心结构

```rust
/// 点击类型
#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
#[repr(i8)]
pub enum ClickType {
    #[default]
    None = 0,
    Left = 1,
    Right = 2,
}

/// 游戏逻辑步骤
#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
#[repr(u8)]
pub enum GameLogicStep {
    #[default]
    None = 0,
    ProcessPlayers,
    ProcessMonsters,
    ProcessObjects,
    ProcessMissiles,
    ProcessItems,
    ProcessTowners,
    ProcessItemsTown,
    ProcessMissilesTown,
}

/// 玩家动作类型
#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
#[repr(u8)]
pub enum PlayerActionType {
    #[default]
    None = 0,
    Walk,
    Spell,
    SpellMonsterTarget,
    SpellPlayerTarget,
    Attack,
    AttackMonsterTarget,
    AttackPlayerTarget,
    OperateObject,
}

/// 游戏ID常量
pub mod game_id {
    pub const DIABLO_FULL: u32 = u32::from_be_bytes(*b"DRTL");
    pub const DIABLO_SPAWN: u32 = u32::from_be_bytes(*b"DSHR");
    pub const HELLFIRE_FULL: u32 = u32::from_be_bytes(*b"HRTL");
    pub const HELLFIRE_SPAWN: u32 = u32::from_be_bytes(*b"HSHR");
}

/// 游戏主状态
pub struct DiabloState {
    /// 地牢种子
    pub dungeon_seeds: [u32; NUMLEVELS],
    /// 关卡种子
    pub level_seeds: [Option<u32>; NUMLEVELS],
    /// 鼠标位置
    pub mouse_position: Point,
    /// 游戏运行结果
    pub run_game_result: bool,
    /// 返回主菜单
    pub return_to_main_menu: bool,
    /// 处理玩家
    pub process_players: bool,
    /// 加载游戏
    pub load_game: bool,
    /// 电影播放中
    pub cine_flag: bool,
    /// 暂停模式 (0=无, 1=暂停, 2=完全暂停)
    pub pause_mode: i32,
    /// 鼠标按下状态
    pub mouse_down: ClickType,
    /// 游戏tick延迟(毫秒)
    pub tick_delay: u16,
    /// 当前游戏逻辑步骤
    pub game_logic_step: GameLogicStep,
    /// 上次玩家动作
    pub last_player_action: PlayerActionType,
}
```
