### 数据结构

```rust
/// 存档文件头
pub struct SaveHeader {
    pub version: u32,
    pub player_name: [u8; 32],
    pub difficulty: Difficulty,
    pub level: u8,
    pub class: PlayerClass,
    pub gold: u32,
    pub timestamp: u64,
}

/// 完整存档
pub struct SaveGame {
    pub header: SaveHeader,
    pub player: PlayerSaveData,
    pub levels: Vec<LevelSaveData>,
    pub quests: QuestProgress,
    pub stash: Vec<Item>,
}

/// 玩家存档数据
pub struct PlayerSaveData {
    pub base_stats: BaseStats,
    pub equipment: [Option<Item>; 7],
    pub inventory: Vec<Item>,
    pub belt: [Option<Item>; 8],
    pub skills: Vec<SkillData>,
    pub spells: Vec<SpellData>,
    pub gold: u32,
}

/// 关卡存档数据
pub struct LevelSaveData {
    pub level_type: DungeonLevelType,
    pub level_num: u8,
    pub tiles: Vec<TileState>,
    pub monsters: Vec<MonsterSaveData>,
    pub objects: Vec<ObjectSaveData>,
    pub items_on_ground: Vec<Item>,
}
```
