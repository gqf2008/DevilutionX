## 集成效果对比

### Before (M14 Day 98 with stubs)

```rust
// ❌ 重复定义 (quests.rs 已有)
pub enum QuestId {
    Rock = 1,
    Mushroom = 2,
    Anvil = 5,
    Betrayer = 13,
}

impl QuestId {
    pub fn is_active(self) -> bool {
        false  // ❌ Always false!
    }
}

// ❌ Quest 对话永远不会触发
let quest_node = manager.get_quest_dialogue(TownerType::Smith, &player);
assert!(quest_node.is_none());  // Always None!
```

### After (M14↔M9 Integration)

```rust
// ✅ 使用 M9 Quest System
use super::quests::{QuestId, QuestManager};

pub fn get_quest_dialogue(
    &self,
    npc: TownerType,
    _player: &Player,
    quest_manager: &QuestManager,  // ✅ Real QuestManager
) -> Option<&DialogueNode> {
    match npc {
        TownerType::Smith => {
            // ✅ 真实 Quest 检查
            if quest_manager.is_active(QuestId::Rock) {
                return Some(&SMITH_QUEST_ROCK_START);
            }
        }
        // ...
    }
    None
}

// ✅ Quest 对话完全功能
let mut quest_manager = QuestManager::new();
quest_manager.set_state(QuestId::Rock, QuestState::Active);
let quest_node = manager.get_quest_dialogue(TownerType::Smith, &player, &quest_manager);
assert!(quest_node.is_some());  // ✅ Quest dialogue returned!
```
