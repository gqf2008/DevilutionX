```rust
//! 游戏内菜单系统
//!
//! C++ Reference: Source/gmenu.cpp

/// 菜单项标志
pub mod MenuFlags {
    pub const ENABLED: u32 = 0x80000000;
    pub const SLIDER: u32 = 0x40000000;
    pub const VALUE_SLIDER: u32 = 0x20000000;
}

/// 菜单项
pub struct MenuItem {
    pub flags: u32,
    pub text: String,
    pub callback: Option<Box<dyn Fn(bool)>>,
    pub value: u32,
    pub max_value: u32,
}

/// 游戏菜单管理器
pub struct GMenu {
    pub items: Vec<MenuItem>,
    pub current_index: usize,
    pub is_active: bool,
    pub is_dragging_slider: bool,
    pub logo_frame: u8,
    pub logo_tick: u32,
}

impl GMenu {
    pub fn new() -> Self;
    pub fn draw(&self, surface: &mut Surface);
    pub fn draw_pause(&self, surface: &mut Surface);
    pub fn up_down(&mut self, is_down: bool);
    pub fn left_right(&mut self, is_right: bool);
    pub fn enter(&mut self);
    pub fn escape(&mut self);
    pub fn get_slider_value(&self, index: usize) -> Option<u32>;
    pub fn set_slider_value(&mut self, index: usize, value: u32);
}
```
