### Day 3-4: 物品生成系统

**文件**: `src/game/items.rs`

```rust
/// 完整物品设置
///
/// **C++ Reference**: `Source/items.cpp:1500-1560`
pub fn setup_all_items(
    player: &Player,
    item: &mut Item,
    idx: ItemIndex,
    seed: u32,
    level: i32,
    uper: i32, // 1=magic, 2=unique, 15=pregen
    only_good: bool,
    pregen: bool,
) {
    // 设置随机种子
    let mut rng = rand::SeedableRng::seed_from_u64(seed as u64);

    // 获取基础属性
    get_item_attrs(item, idx, level);

    // 根据 uper 决定物品品质
    let quality = determine_quality(&mut rng, uper, pregen);
    item.quality = quality;

    match quality {
        ItemQuality::Normal => {
            // 普通物品无词缀
        }
        ItemQuality::Magic => {
            // 魔法物品应用词缀
            get_item_bonus(player, item, level / 2, level, only_good, true);
        }
        ItemQuality::Unique => {
            // 唯一物品特殊处理
            apply_unique_item(item, idx, seed);
        }
        _ => {}
    }

    // 计算最终价值
    calculate_item_value(item);
}
```

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_get_item_attrs_sword() {
        let mut item = Item::default();
        get_item_attrs(&mut item, ItemIndex::ShortSword, 5);

        assert_eq!(item.item_type, ItemType::Sword);
        assert!(item.min_damage > 0);
        assert!(item.max_damage >= item.min_damage);
    }

    #[test]
    fn test_get_item_bonus_magic() {
        let player = Player::default();
        let mut item = Item::default();
        get_item_attrs(&mut item, ItemIndex::LongSword, 10);

        get_item_bonus(&player, &mut item, 5, 10, false, true);

        // 应该有词缀
        assert!(item.prefix_id.is_some() || item.suffix_id.is_some());
    }

    #[test]
    fn test_setup_all_items_unique() {
        let player = Player::default();
        let mut item = Item::default();

        setup_all_items(&player, &mut item, ItemIndex::LongSword, 12345, 15, 2, false, false);

        assert_eq!(item.quality, ItemQuality::Unique);
    }
}
