### Rust 设计

```rust
/// 从中心向外爬行遍历
///
/// # Arguments
/// * `radius` - 遍历半径
/// * `callback` - 每个位置的回调函数，返回false停止遍历
///
/// # Returns
/// 如果遍历完成返回true，被中断返回false
pub fn do_crawl<F>(radius: u32, callback: F) -> bool
where
    F: FnMut(Displacement) -> bool;

pub fn do_crawl_range<F>(min_radius: u32, max_radius: u32, callback: F) -> bool
where
    F: FnMut(Displacement) -> bool;
```
