```rust
/// 怪物攻击玩家
pub fn monster_attack_player(
    monster: &Monster,
    player: &mut Player,
    rng: &mut impl Rng,
) -> AttackResult {
    // 1. 计算命中率
    let to_hit_chance = calculate_monster_to_hit(monster, player);

    // 2. 命中判定
    if rng.gen_range(0..100) >= to_hit_chance {
        return AttackResult::Miss;
    }

    // 3. 计算伤害
    let base_damage = rng.gen_range(monster.min_damage..=monster.max_damage);
    let armor_reduction = player.get_armor_class() / 2;
    let final_damage = base_damage.saturating_sub(armor_reduction).max(1);

    // 4. 应用伤害 (64x fixed-point)
    player.modify_hp(-(final_damage * 64));

    AttackResult::Hit { damage: final_damage }
}

/// 计算怪物命中率 (C++ formula from M_TryH2HHit)
fn calculate_monster_to_hit(monster: &Monster, player: &Player) -> i32 {
    let base_chance = monster.to_hit;
    let ac_modifier = player.get_armor_class();

    // C++ formula: 30 + monster.to_hit - player.ac
    let hit_chance = 30 + base_chance - ac_modifier;

    hit_chance.clamp(5, 95) // 5%-95% 命中率
}
```

#### Task 2: Player 攻击 Monster

**C++ 参考**: `Source/player.cpp` - `PM_DoAttack()`

```rust
/// 玩家攻击怪物
pub fn player_attack_monster(
    player: &Player,
    monster: &mut Monster,
    rng: &mut impl Rng,
) -> AttackResult {
    // 1. 计算命中率
    let to_hit_chance = calculate_player_to_hit(player, monster);

    // 2. 命中判定
    if rng.gen_range(0..100) >= to_hit_chance {
        return AttackResult::Miss;
    }

    // 3. 计算伤害 (武器 + 力量加成)
    let base_damage = player.get_melee_damage(rng);
    let armor_reduction = monster.armor_class / 4;
    let final_damage = base_damage.saturating_sub(armor_reduction).max(1);

    // 4. 应用伤害 (64x fixed-point)
    monster.modify_hp(-(final_damage * 64));

    // 5. 检查死亡
    if monster.hit_points <= 0 {
        monster.mode = MonsterMode::Dying;
        return AttackResult::Kill { damage: final_damage };
    }

    AttackResult::Hit { damage: final_damage }
}

/// 计算玩家命中率 (C++ formula)
fn calculate_player_to_hit(player: &Player, monster: &Monster) -> i32 {
    let level_bonus = player.level / 2;
    let dex_bonus = player.dexterity / 4;
    let weapon_to_hit = player.get_weapon_to_hit();

    // C++ formula: 50 + level/2 + dex/4 + weapon_to_hit - monster.ac
    let hit_chance = 50 + level_bonus + dex_bonus + weapon_to_hit - monster.armor_class;

    hit_chance.clamp(5, 95)
}
```

```rust
/// 怪物尝试打开门
pub fn monster_open_door(
    monster: &Monster,
    door: &mut Object,
) -> bool {
    // 1. 检查怪物能力
    if !monster.flags.contains(MonsterFlags::CAN_OPEN_DOOR) {
        return false;
    }

    // 2. 检查门状态
    if !matches!(door.id, ObjectId::DoorL1 | ObjectId::DoorL2 | ObjectId::DoorL3) {
        return false;
    }

    if door.is_open {
        return false; // 已经开启
    }

    // 3. 打开门
    door.is_open = true;
    door.solid = false;
    door.animation_frame = 0;

    true
}
```

```rust
pub fn monster_break_object(
    monster: &Monster,
    object: &mut Object,
) -> bool {
    // 1. 检查对象可破坏
    if !object.breakable {
        return false;
    }

    // 2. 某些怪物无法破坏 (如史莱姆)
    if monster.mtype == MonsterType::Slime {
        return false;
    }

    // 3. 破坏对象
    object.active = false;
    object.solid = false;

    // 4. 可能生成战利品 (未来实现)
    true
}
```

#### Task 2: Update Loop

**C++ 参考**: `Source/diablo.cpp` - `GameLoop()`

```rust
impl GameState {
    /// 主游戏循环 (每帧调用)
    pub fn update(&mut self, delta_time: u32, rng: &mut impl Rng) {
        if self.paused {
            return;
        }

        self.game_tick += 1;

        // 1. 更新怪物 AI 和移动
        self.update_monsters(delta_time, rng);

        // 2. 更新对象动画
        self.objects.update_animations(delta_time);

        // 3. 检查怪物攻击玩家
        self.check_monster_attacks(rng);

        // 4. 清理死亡怪物
        self.cleanup_dead_monsters();
    }

    /// 更新所有怪物 AI
    fn update_monsters(&mut self, delta_time: u32, rng: &mut impl Rng) {
        for i in 0..self.monsters.count() {
            if let Some(monster) = self.monsters.get_mut(i) {
                if !monster.active {
                    continue;
                }

                // AI 决策
                let ai = monster.get_ai();
                let action = ai.update(monster, self.player.position, rng);

                // 执行 AI 动作
                apply_ai_action(monster, action, &self.player, rng);

                // 检查与对象交互
                if monster.flags.contains(MonsterFlags::CAN_OPEN_DOOR) {
                    if let Some(door_idx) = self.objects.find_door_at(monster.position) {
                        if let Some(door) = self.objects.get_mut(door_idx) {
                            monster_open_door(monster, door);
                        }
                    }
                }
            }
        }
    }

    /// 检查怪物攻击范围内的玩家
    fn check_monster_attacks(&mut self, rng: &mut impl Rng) {
        let player_pos = self.player.position;

        for i in 0..self.monsters.count() {
            if let Some(monster) = self.monsters.get(i) {
                if !monster.active || monster.mode != MonsterMode::Attacking {
                    continue;
                }

                // 检查是否在攻击范围内 (1 格)
                let distance = walking_distance(monster.position, player_pos);
                if distance <= 1 {
                    let result = monster_attack_player(monster, &mut self.player, rng);

                    if let AttackResult::Hit { damage } = result {
                        // 可以添加日志或音效触发
                        println!("Monster hit player for {} damage!", damage);
                    }
                }
            }
        }
    }

    /// 清理死亡怪物
    fn cleanup_dead_monsters(&mut self) {
        let mut removed = 0;
        for i in 0..self.monsters.count() {
            if let Some(monster) = self.monsters.get(i) {
                if monster.mode == MonsterMode::Dead {
                    self.monsters.remove(i - removed);
                    self.kills += 1;
                    removed += 1;
                }
            }
        }
    }
}
```

#### Task 3: 玩家动作接口

```rust
impl GameState {
    /// 玩家移动
    pub fn player_move(&mut self, direction: Direction) -> bool {
        let new_pos = self.player.position + direction.offset();

        // 检查对象阻挡
        if let Some(obj) = self.objects.get_object_at(new_pos) {
            if obj.solid {
                return false;
            }
        }

        // 检查怪物阻挡
        if self.monsters.find_monster_at(new_pos).is_some() {
            return false;
        }

        self.player.position = new_pos;
        true
    }

    /// 玩家攻击
    pub fn player_attack(&mut self, target_pos: Point, rng: &mut impl Rng) -> Option<AttackResult> {
        if let Some(monster_idx) = self.monsters.find_monster_at(target_pos) {
            if let Some(monster) = self.monsters.get_mut(monster_idx) {
                let result = player_attack_monster(&self.player, monster, rng);
                return Some(result);
            }
        }
        None
    }

    /// 玩家交互对象
    pub fn player_interact(&mut self, target_pos: Point) -> bool {
        if let Some(obj_idx) = self.objects.find_object_at(target_pos) {
            if let Some(obj) = self.objects.get_mut(obj_idx) {
                // 调用 objects.rs 的交互逻辑
                return obj.interact(&mut self.player);
            }
        }
        false
    }
}
```

```rust
pub fn update(&mut self, rng: &mut impl Rng) {
    if self.is_town {
        return; // 城镇跳过 Monster/Object 处理
    }

    self.process_player(rng);    // 玩家 HP/Mana 恢复
    self.process_monsters(rng);  // 怪物 HP 恢复 + AI
    self.process_objects();      // 对象动画更新

    self.game_tick += 1;
}
```

### 技术亮点

1. **MonsterManager 迭代器模式**: 使用 `iter()` 而非索引访问，避免 borrow checker 冲突
2. **类型安全转换**: `_p_level` (u8) → i32 转换用于 HP 计算
3. **ID 预收集模式**: 先收集 monster IDs，再逐个处理，避免同时可变借用
4. **Town 优化**: `is_town` 标志跳过 Monster/Object 处理，节省性能

### 解决的问题

#### 问题 1: MonsterManager API 不匹配
- **错误**: 调用不存在的 `get_active_monster_id()` 方法
- **原因**: MonsterManager 使用迭代器模式，不提供索引枚举
- **解决**: 改用 `iter()` 并预收集 ID:
  ```rust
  let monster_ids: Vec<usize> = self.monster_manager.iter().map(|(id, _)| id).collect();
  ```

#### 问题 2: 类型转换错误
- **错误**: `cannot add u8 to i32` (Player._p_level 是 u8)
- **解决**: 显式转换: `(self.player._p_level as i32) / 4 + 1`

#### 问题 3: Borrow Checker 冲突
- **错误**: 在遍历 monsters 时无法可变借用 player
- **解决**: 先收集 monster IDs，再在独立循环中处理战斗
