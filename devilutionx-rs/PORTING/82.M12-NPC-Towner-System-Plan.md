        return Quests[Q_BUTCHER]._qactive != QUEST_NOTAVAIL

        return gbIsHellfire && sgGameInitInfo.bCowQuest == 0

        return gbIsHellfire && sgGameInitInfo.bCowQuest != 0;

2. **TownerAvailability 可用性检查**
   ```rust
   pub enum TownerAvailability {
       Always,                          // Default NPCs
       QuestActive(QuestId),            // DeadGuy (Q_BUTCHER)
       HellfireOnly,                    // Farmer, Girl, CowFarmer
       ComplexCondition,                // Multi-condition
   }

   pub fn is_towner_available(
       towner_type: TownerType,
       game_mode: GameMode,
       quests: &[QuestState],
       player_visited_levels: &[bool],
   ) -> bool {
       match towner_type {
           TownerType::DeadGuy => {
               // Butcher quest: active but not done
               quests[QuestId::Butcher as usize].is_active()
                   && !quests[QuestId::Butcher as usize].is_done()
           }
           TownerType::Farmer => {
               // Hellfire + Cow quest not started + Hive not done
               game_mode == GameMode::Hellfire
                   && !is_cow_quest_started()
                   && quests[QuestId::Farmer as usize] != QuestState::HiveDone
           }
           TownerType::CowFarmer => {
               // Hellfire + Cow quest active
               game_mode == GameMode::Hellfire && is_cow_quest_started()
           }
           TownerType::Girl => {
               // Hellfire + Theo quest + visited Hell level
               game_mode == GameMode::Hellfire
                   && is_theo_quest_active()
                   && player_visited_levels[17]
                   && !quests[QuestId::Girl as usize].is_done()
           }
           _ => true, // Always available
       }
   }
   ```

   ```rust
   pub struct TownerFactory;

   impl TownerFactory {
       pub fn create_towner(
           towner_type: TownerType,
           position: Option<Point>,
       ) -> Towner {
           let data = Self::get_data(towner_type);
           let pos = position.unwrap_or(data.default_position);
           let gossip = Self::random_gossip(data.gossip_texts);

           Towner {
               towner_type,
               name: data.name.to_string(),
               position: pos,
               gossip,
               anim_width: data.anim_width,
               anim_delay: data.anim_delay,
               anim_cnt: 0,
               anim_len: data.anim_frames,
               anim_frame: 0,
               anim_frame_cnt: 0,
           }
       }

       pub fn get_data(towner_type: TownerType) -> &'static TownerDataEntry {
           &TOWNER_DATA[towner_type as usize]
       }

       fn random_gossip(texts: &[SpeechId]) -> SpeechId {
           // Simple random selection (deterministic for testing)
           if texts.is_empty() {
               SpeechId::None
           } else {
               texts[0] // Simplified for now
           }
       }
   }
   ```

3. **init_towners() 全局初始化**
   ```rust
   pub fn init_towners(
       game_mode: GameMode,
       quests: &[QuestState],
       player_visited_levels: &[bool],
   ) -> TownerRegistry {
       let mut registry = TownerRegistry::new();

       for towner_type in [
           TownerType::Smith,
           TownerType::Healer,
           // ... all 13 types
       ] {
           if is_towner_available(
               towner_type,
               game_mode,
               quests,
               player_visited_levels,
           ) {
               let towner = TownerFactory::create_towner(towner_type, None);
               registry.add_towner(towner);
           }
       }

       registry
   }
   ```
