fn place_l4_penta(&mut self, dungeon: &mut Dungeon, diablo_quest_pos: (usize, usize)) {
    let l4_penta = miniset_l4_penta();

    // Check if gate is open
    let is_gate_open = self.use_multiplayer_quests()
        || self.is_quest_done(QuestType::Diablo);

    if !is_gate_open {
        // Place pentagram at Diablo quest position
        let (qx, qy) = diablo_quest_pos;
        for dy in 0..l4_penta.height {
            for dx in 0..l4_penta.width {
                if qx + dx < DMAXX && qy + dy < DMAXY {
                    let tile = l4_penta.replace[dy][dx];
                    if tile != 0 {
                        dungeon.tiles[qx + dx][qy + dy] = tile;
                    }
                }
            }
        }
    }
}
```

fn place_dun_tiles(&mut self, dungeon: &mut Dungeon,
                   data: &[u16], pos: (usize, usize), floor: u8) {
    // 放置.dun瓦片
    let width = data[0] as usize;
    let height = data[1] as usize;
    let mut idx = 2;

    for y in 0..height {
        for x in 0..width {
            if pos.0 + x < DMAXX && pos.1 + y < DMAXY {
                let tile = data[idx] as u8;
                if tile != floor {
                    dungeon.tiles[pos.0 + x][pos.1 + y] = tile;
                }
            }
            idx += 1;
        }
    }
}
```
