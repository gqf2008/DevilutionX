```rust
use std::fs::File;
use std::io::Read;

/// Load .dun file and place tiles in dungeon
fn load_dun_file(&mut self, filename: &str, dungeon: &mut Dungeon, x: i32, y: i32) -> bool {
    // Open file
    let mut file = match File::open(filename) {
        Ok(f) => f,
        Err(_) => return false,
    };

    // Read header (4 bytes)
    let mut header = [0u8; 4];
    if file.read_exact(&mut header).is_err() {
        return false;
    }

    // Parse width/height (little-endian)
    let width = u16::from_le_bytes([header[0], header[1]]) as usize;
    let height = u16::from_le_bytes([header[2], header[3]]) as usize;

    // Read tile data
    let tile_count = width * height;
    let mut tile_data = vec![0u16; tile_count];
    let mut buffer = vec![0u8; tile_count * 2];

    if file.read_exact(&mut buffer).is_err() {
        return false;
    }

    // Convert to u16 array
    for i in 0..tile_count {
        tile_data[i] = u16::from_le_bytes([buffer[i*2], buffer[i*2+1]]);
    }

    // Place tiles in dungeon
    place_dun_tiles(dungeon, &tile_data, width, height, x, y);

    true
}

/// Place .dun tile data into dungeon
fn place_dun_tiles(dungeon: &mut Dungeon, tiles: &[u16], width: usize, height: usize, x: i32, y: i32) {
    for j in 0..height {
        for i in 0..width {
            let dx = x + i as i32;
            let dy = y + j as i32;

            if dx >= 0 && dx < DMAXX as i32 && dy >= 0 && dy < DMAXY as i32 {
                let tile_id = tiles[j * width + i];
                dungeon.tiles[dx as usize][dy as usize] = tile_id as u8;
            }
        }
    }
}
```

### .dun File Format

**Binary Structure**:
```
Offset | Size | Type | Description
-------|------|------|-------------
0x00   | 2    | u16  | Width (little-endian)
0x02   | 2    | u16  | Height (little-endian)
0x04   | W*H*2| u16[]| Tile data (little-endian)
```

**Example**:
```
File: diab1.dun
Header: [06 00 06 00] = 6×6 tiles
Data: [01 00 02 00 ...] = tile IDs 1, 2, ...
```
