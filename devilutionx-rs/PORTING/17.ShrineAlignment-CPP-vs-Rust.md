**C++ Logic:**
```cpp
void OperateShrineCostOfWisdom(Player &player, SpellID spellId, diablo_message message)
{
    // 1. Grant spell to player
    player._pMemSpells |= GetSpellBitmask(spellId);
    
    // 2. Increase spell level by 2 (max 15)
    if (curSpellLevel < MaxSpellLevel) {
        player._pSplLvl[spellId] = min(curSpellLevel + 2, MaxSpellLevel);
    }
    
    // 3. Reduce max mana by 10%
    uint32_t t = player._pMaxManaBase / 10;
    player._pManaBase -= t;
    player._pMana -= t;
    player._pMaxMana -= t;
    player._pMaxManaBase -= t;
}
```
