/// 计算居中偏移
pub fn calculate_centered_offset(window: Size, output: Size) -> Point {
    Point::new(
        (window.width as i32 - output.width as i32) / 2,
        (window.height as i32 - output.height as i32) / 2,
    )
}

/// 计算整数缩放
pub fn calculate_integer_scale(window: Size, base: Size) -> (u32, Size) {
    let scale_x = window.width / base.width;
    let scale_y = window.height / base.height;
    let scale = scale_x.min(scale_y).max(1);
    let output = Size::new(base.width * scale, base.height * scale);
    (scale, output)
}
