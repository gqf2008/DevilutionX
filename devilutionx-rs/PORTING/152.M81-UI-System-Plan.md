pub struct Button {
    text: String,
    rect: Rect,
    state: ButtonState,
    callback: Option<Box<dyn FnMut()>>,
}

pub enum ButtonState {
    Normal,
    Hovered,
    Pressed,
    Disabled,
}

impl Widget for Button {
    fn draw(&self, ctx: &mut UiContext) {
        // 绘制按钮背景、边框、文本
        match self.state {
            ButtonState::Normal => /* 正常样式 */,
            ButtonState::Hovered => /* 高亮样式 */,
            ButtonState::Pressed => /* 按下样式 */,
            ButtonState::Disabled => /* 禁用样式 */,
        }
    }

    fn handle_input(&mut self, input: &InputSystem) -> WidgetEvent {
        // 检测鼠标悬停、点击
        if self.rect.contains(input.mouse_pos()) {
            if input.is_mouse_button_pressed(MouseButton::Left) {
                self.state = ButtonState::Pressed;
                return WidgetEvent::Clicked;
            }
            self.state = ButtonState::Hovered;
        } else {
            self.state = ButtonState::Normal;
        }
        WidgetEvent::None
    }
}

pub enum MainMenuResult {
    SinglePlayer,
    MultiPlayer,
    Settings,
    Exit,
}

pub struct MainMenuDialog {
    title_text: String,
    buttons: Vec<Button>,
    background: Option<ClxSprite>,
    selected_index: usize,
}

impl MainMenuDialog {
    pub fn new(ctx: &UiContext) -> Self {
        let mut dialog = Self {
            title_text: "Diablo".to_string(),
            buttons: vec![
                Button::new("Single Player", Rect::new(300, 200, 200, 50)),
                Button::new("Multi Player", Rect::new(300, 270, 200, 50)),
                Button::new("Settings", Rect::new(300, 340, 200, 50)),
                Button::new("Exit Diablo", Rect::new(300, 410, 200, 50)),
            ],
            background: None,
            selected_index: 0,
        };

        // 加载背景图片
        if let Ok(bg) = ctx.resources().load_sprite("ui_art/mainmenu.pcx") {
            dialog.background = Some(bg);
        }

        dialog
    }

    pub fn show(&mut self, ctx: &mut UiContext) -> MainMenuResult {
        loop {
            let input = ctx.update_input();

            // 键盘导航
            if input.is_action_pressed(GameAction::MenuDown) {
                self.selected_index = (self.selected_index + 1) % self.buttons.len();
            }
            if input.is_action_pressed(GameAction::MenuUp) {
                self.selected_index = (self.selected_index + self.buttons.len() - 1) % self.buttons.len();
            }
            if input.is_action_pressed(GameAction::MenuSelect) {
                return self.get_selection();
            }

            // 鼠标点击
            for (i, button) in self.buttons.iter_mut().enumerate() {
                if let WidgetEvent::Clicked = button.handle_input(&input) {
                    self.selected_index = i;
                    return self.get_selection();
                }
            }

            // 渲染
            ctx.window_mut().clear(Color::RGB(0, 0, 0));

            if let Some(ref bg) = self.background {
                ctx.draw_sprite(bg, 0, 0);
            }

            ctx.draw_centered_text(&self.title_text, 400, 100, Color::RGB(255, 255, 0));

            for (i, button) in self.buttons.iter().enumerate() {
                button.set_focused(i == self.selected_index);
                button.draw(ctx);
            }

            ctx.window_mut().present();

            if input.is_action_pressed(GameAction::Quit) {
                return MainMenuResult::Exit;
            }
        }
    }

    fn get_selection(&self) -> MainMenuResult {
        match self.selected_index {
            0 => MainMenuResult::SinglePlayer,
            1 => MainMenuResult::MultiPlayer,
            2 => MainMenuResult::Settings,
            3 => MainMenuResult::Exit,
            _ => MainMenuResult::Exit,
        }
    }
}

impl HeroSelectDialog {
    pub fn new(ctx: &UiContext) -> Result<Self> {
        // 加载现有角色列表
        let hero_list = load_saved_heroes()?;

        Ok(Self {
            hero_list,
            selected_index: 0,
            buttons: vec![
                Button::new("Create Hero", Rect::new(50, 500, 150, 40)),
                Button::new("Load Hero", Rect::new(220, 500, 150, 40)),
                Button::new("Delete Hero", Rect::new(390, 500, 150, 40)),
                Button::new("Back", Rect::new(560, 500, 150, 40)),
            ],
        })
    }

    pub fn show(&mut self, ctx: &mut UiContext) -> HeroSelectResult {
        loop {
            let input = ctx.update_input();

            // 键盘导航 (上下选择角色)
            if input.is_action_pressed(GameAction::MenuDown) {
                self.selected_index = (self.selected_index + 1).min(self.hero_list.len());
            }
            if input.is_action_pressed(GameAction::MenuUp) {
                self.selected_index = self.selected_index.saturating_sub(1);
            }

            // 渲染
            ctx.window_mut().clear(Color::RGB(0, 0, 0));

            // 绘制标题
            ctx.draw_centered_text("Select Hero", 400, 50, Color::RGB(255, 255, 0));

            // 绘制角色列表
            for (i, hero) in self.hero_list.iter().enumerate() {
                let y = 150 + i as i32 * 60;
                let color = if i == self.selected_index {
                    Color::RGB(255, 255, 0) // 高亮
                } else {
                    Color::RGB(180, 180, 180)
                };

                let info = format!("{} - Lv.{} {:?}", hero.name, hero.level, hero.class);
                ctx.draw_text(&info, 100, y, color);
            }

            // 绘制按钮
            for button in &self.buttons {
                button.draw(ctx);
            }

            ctx.window_mut().present();

            // 处理按钮点击
            if self.buttons[0].handle_input(&input) == WidgetEvent::Clicked {
                return HeroSelectResult::CreateNew;
            }
            if self.buttons[1].handle_input(&input) == WidgetEvent::Clicked {
                if self.selected_index < self.hero_list.len() {
                    return HeroSelectResult::LoadHero(self.hero_list[self.selected_index].clone());
                }
            }
            if self.buttons[3].handle_input(&input) == WidgetEvent::Clicked {
                return HeroSelectResult::Back;
            }
        }
    }
}

// 辅助函数: 加载保存的角色
fn load_saved_heroes() -> Result<Vec<HeroInfo>> {
    let save_dir = std::env::current_dir()?.join("saves");
    let mut heroes = Vec::new();

    if save_dir.exists() {
        for entry in std::fs::read_dir(save_dir)? {
            let path = entry?.path();
            if path.extension().map_or(false, |ext| ext == "hero") {
                if let Ok(data) = std::fs::read_to_string(&path) {
                    if let Ok(hero) = serde_json::from_str::<HeroInfo>(&data) {
                        heroes.push(hero);
                    }
                }
            }
        }
    }

    Ok(heroes)
}

use std::ffi::CString;
use std::path::Path;

// 使用 3rdParty/libmpq 的Rust绑定
pub struct MpqArchive {
    handle: *mut libmpq::mpq_archive_t,
}

impl MpqArchive {
    pub fn open<P: AsRef<Path>>(path: P) -> Result<Self> {
        let path_str = path.as_ref().to_str().ok_or("Invalid path")?;
        let c_path = CString::new(path_str)?;

        let mut handle = std::ptr::null_mut();
        unsafe {
            let result = libmpq::libmpq__archive_open(&mut handle, c_path.as_ptr(), -1);
            if result != 0 {
                return Err(format!("Failed to open MPQ: {}", path_str).into());
            }
        }

        Ok(Self { handle })
    }

    pub fn read_file(&self, filename: &str) -> Result<Vec<u8>> {
        let c_filename = CString::new(filename)?;

        unsafe {
            // 获取文件大小
            let mut file_size = 0;
            libmpq::libmpq__file_size_unpacked(self.handle, c_filename.as_ptr(), &mut file_size);

            // 读取文件
            let mut buffer = vec![0u8; file_size as usize];
            let mut bytes_read = 0;
            let result = libmpq::libmpq__file_read(
                self.handle,
                c_filename.as_ptr(),
                buffer.as_mut_ptr(),
                file_size,
                &mut bytes_read,
            );

            if result != 0 {
                return Err(format!("Failed to read file: {}", filename).into());
            }

            Ok(buffer)
        }
    }

    }
}

// 全局MPQ管理器
pub struct MpqManager {
    archives: Vec<MpqArchive>,
}

impl MpqManager {
    pub fn new() -> Self {
        Self { archives: Vec::new() }
    }

    pub fn load_core_archives(&mut self) -> Result<()> {
        self.archives.push(MpqArchive::open("diabdat.mpq")?);
        self.archives.push(MpqArchive::open("devilutionx.mpq")?);
        self.archives.push(MpqArchive::open("fonts.mpq")?);
        Ok(())
    }

    pub fn read_file(&self, filename: &str) -> Result<Vec<u8>> {
        // 从多个档案中查找文件 (优先级)
        for archive in self.archives.iter().rev() {
            if let Ok(data) = archive.read_file(filename) {
                return Ok(data);
            }
        }
        Err(format!("File not found: {}", filename).into())
    }
}

        // 4. 播放Logo (可选)
        // self.show_splash()?;

        // 5. 主菜单循环
        loop {
            let menu_result = self.show_main_menu()?;

            match menu_result {
                MainMenuResult::SinglePlayer => {
                    if let Some(hero) = self.single_player_flow()? {
                        self.start_game(hero)?;
                    }
                }
                MainMenuResult::MultiPlayer => {
                    // TODO: 多人游戏
                }
                MainMenuResult::Settings => {
                    // TODO: 设置界面
                }
                MainMenuResult::Exit => break,
            }
        }

        Ok(())
    }

    fn show_main_menu(&mut self) -> Result<MainMenuResult> {
        let mut ctx = UiContext::new(self.window.clone(), self.resources.clone());
        let mut dialog = MainMenuDialog::new(&ctx);
        Ok(dialog.show(&mut ctx))
    }

    fn single_player_flow(&mut self) -> Result<Option<HeroInfo>> {
        let mut ctx = UiContext::new(self.window.clone(), self.resources.clone());

        // 角色选择
        let mut select_dialog = HeroSelectDialog::new(&ctx)?;

        match select_dialog.show(&mut ctx) {
            HeroSelectResult::CreateNew => {
                // 创建新角色
                let mut create_dialog = HeroCreateDialog::new(&ctx);
                Ok(create_dialog.show(&mut ctx))
            }
            HeroSelectResult::LoadHero(hero) => {
                // 加载现有角色
                Ok(Some(hero))
            }
            HeroSelectResult::Back => Ok(None),
        }
    }

    fn start_game(&mut self, hero: HeroInfo) -> Result<()> {
        println!("Starting game with hero: {}", hero.name);

        // 初始化游戏状态
        let mut game_state = GameState::new(hero);

        // 运行游戏循环
        run_game_loop(&mut game_state, &mut self.window)?;

        Ok(())
    }
}
