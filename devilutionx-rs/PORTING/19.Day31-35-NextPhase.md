// Monster attacks Player
pub fn monster_attack_player(
    monster: &mut Monster,
    player: &mut Player,
    rng: &mut impl Rng,
) -> CombatResult {
    // 1. Calculate to-hit
    let to_hit = monster.to_hit - player.get_armor_class();
    let roll = rng.random_range(0..100);

    if roll < to_hit {
        // Hit!
        let damage = rng.random_range(monster.min_damage..=monster.max_damage);
        let actual_damage = apply_player_resistances(player, damage, monster.attack_type);
        player.modify_hp(-actual_damage);

        CombatResult::Hit { damage: actual_damage }
    } else {
        CombatResult::Miss
    }
}

// Player attacks Monster
pub fn player_attack_monster(
    player: &Player,
    monster: &mut Monster,
    rng: &mut impl Rng,
) -> CombatResult {
    // 1. Calculate to-hit
    let to_hit = player.get_to_hit() - monster.armor_class;
    let roll = rng.random_range(0..100);

    if roll < to_hit {
        // Hit!
        let damage = calculate_player_damage(player, rng);
        monster.hit_points -= damage;

        if monster.hit_points <= 0 {
            monster.mode = MonsterMode::Dying;
            CombatResult::Kill { damage, exp: monster.mexp }
        } else {
            monster.mode = MonsterMode::GotHit;
            CombatResult::Hit { damage }
        }
    } else {
        CombatResult::Miss
    }
}
```

### Task 4: Monster AI Framework (Day 34-35)

**C++ Reference**: `Source/monster.cpp` - AI functions

pub trait MonsterAI {
    fn update(&self, monster: &mut Monster, player: &Player, delta: u32);
    fn should_attack(&self, monster: &Monster, player: &Player) -> bool;
    fn choose_target(&self, monster: &Monster, player: &Player) -> Point;
}

// Example: Zombie AI
pub struct ZombieAI;

impl MonsterAI for ZombieAI {
    fn update(&self, monster: &mut Monster, player: &Player, delta: u32) {
        match monster.mode {
            MonsterMode::Standing => {
                // Check if player in range
                if self.should_attack(monster, player) {
                    monster.mode = MonsterMode::Attacking;
                    monster.enemy = player.position;
                }
            }
            MonsterMode::Attacking => {
                // Move towards player or attack
                let dist = distance(monster.position, player.position);
                if dist <= 1 {
                    // Melee attack
                    monster_attack_player(monster, player, &mut rng);
                } else {
                    // Move closer
                    move_monster_towards(monster, player.position);
                }
            }
            _ => {}
        }
    }

    fn should_attack(&self, monster: &Monster, player: &Player) -> bool {
        let dist = distance(monster.position, player.position);
        dist < 10 // Zombie aggro range
    }

    fn choose_target(&self, monster: &Monster, player: &Player) -> Point {
        player.position
    }
}
```

## 📝 Example Integration

**Game Loop Integration**:
```rust
// In main game loop
pub struct GameState {
    player: Player,
    monsters: MonsterManager,
    objects: ObjectManager,
}

impl GameState {
    pub fn update(&mut self, delta: u32) {
        // 1. Update monsters (AI + movement)
        for i in 0..self.monsters.active_count {
            if let Some(monster) = self.monsters.get_monster_mut(i) {
                let ai = get_ai_for_type(monster.ai_type);
                ai.update(monster, &self.player, delta);
            }
        }

        // 2. Update objects
        self.objects.process_animations(delta);

        // 3. Check player input
        if player_attacks() {
            if let Some(target) = self.monsters.find_monster_at(player_target_pos) {
                if let Some(monster) = self.monsters.get_monster_mut(target) {
                    player_attack_monster(&self.player, monster, &mut rng);
                }
            }
        }

        // 4. Cleanup dead monsters
        self.monsters.cleanup_dead();
    }
}
```
