### 1. River 完整实现 (river)

**C++ 等价**: `River()` (lines 1013-1213)

**复杂度**:
- 路径追踪: 4方向移动 + nodir/nodir2 双重阻塞
- 拐角检测: 8种拐角模式 (tiles 19-22)
- 端点检测: 4种边缘端点 (tiles 23-24/38-43)
- 桥梁放置: 随机替换河流段为桥梁 (tiles 44-45)

**关键逻辑**:

```rust
fn river(&mut self, dungeon: &mut Dungeon) {
    let mut river_count = 0;
    let mut tries = 0;
    let mut pdir = -1i32; // Previous direction

    while tries < 200 && river_count < 4 {
        tries += 1;

        // 1. Find starting point (tiles 25-28: directional edge tiles)
        let mut rx = 0usize;
        let mut ry = 0usize;
        let mut i = 0;

        while i < 100 {
            rx = self.random_range(0, DMAXX);
            ry = self.random_range(0, DMAXY);
            i += 1;

            // Scan horizontally if needed (matches C++ bugfix)
            while ry < DMAXY && (dungeon.tiles[rx][ry] < 25 || dungeon.tiles[rx][ry] > 28) {
                rx += 1;
                if rx >= DMAXX {
                    rx = 0;
                    ry += 1;
                }
            }

            if ry < DMAXY && dungeon.tiles[rx][ry] >= 25 && dungeon.tiles[rx][ry] <= 28 {
                break;
            }
        }

        if ry >= DMAXY || i >= 100 {
            continue;
        }

        // 2. Initialize river path based on starting tile
        let mut river_x = [0usize; 100];
        let mut river_y = [0usize; 100];
        let mut river_tile = [0u8; 100];

        let (mut dir, nodir) = match dungeon.tiles[rx][ry] {
            25 => (3, 2), // West start → East blocked
            26 => (0, 1), // North start → South blocked
            27 => (1, 0), // South start → North blocked
            28 => (2, 3), // East start → West blocked
            _ => continue,
        };

        // Set starting endpoint tile
        river_tile[0] = match dungeon.tiles[rx][ry] {
            25 => 40, 26 => 38, 27 => 41, 28 => 39, _ => 0,
        };
        river_x[0] = rx;
        river_y[0] = ry;
        let mut riveramt = 1;

        let mut nodir2 = 4; // Secondary blocked direction
        let mut dircheck = 0;

        // 3. Trace river path
        while dircheck < 4 && riveramt < 100 {
            let px = rx;
            let py = ry;

            if dircheck == 0 {
                dir = self.random_range(0, 4) as i32;
            } else {
                dir = (dir + 1) & 3; // Rotate direction
            }
            dircheck += 1;

            // Skip blocked directions
            while dir == nodir || dir == nodir2 {
                dir = (dir + 1) & 3;
                dircheck += 1;
            }

            // Move in direction
            match dir {
                0 if ry > 0 => ry -= 1,         // North
                1 if ry < DMAXY - 1 => ry += 1, // South
                2 if rx < DMAXX - 1 => rx += 1, // East
                3 if rx > 0 => rx -= 1,         // West
                _ => {}
            }

            // Check if we hit floor tile
            if dungeon.tiles[rx][ry] == 7 {
                dircheck = 0;

                // Choose straight river tile based on direction
                if dir < 2 {
                    // Vertical (North/South)
                    river_tile[riveramt] = if self.flip_coin() { 17 } else { 18 };
                } else {
                    // Horizontal (East/West)
                    river_tile[riveramt] = if self.flip_coin() { 15 } else { 16 };
                }

                river_x[riveramt] = rx;
                river_y[riveramt] = ry;
                riveramt += 1;

                // 4. Detect corners and update previous tile
                if (dir == 0 && pdir == 2) || (dir == 3 && pdir == 1) {
                    // North←East or West←South
                    if riveramt > 2 {
                        river_tile[riveramt - 2] = 22;
                    }
                    nodir2 = if dir == 0 { 1 } else { 2 };
                }
                if (dir == 0 && pdir == 3) || (dir == 2 && pdir == 1) {
                    // North←West or East←South
                    if riveramt > 2 {
                        river_tile[riveramt - 2] = 21;
                    }
                    nodir2 = if dir == 0 { 1 } else { 3 };
                }
                if (dir == 1 && pdir == 2) || (dir == 3 && pdir == 0) {
                    // South←East or West←North
                    if riveramt > 2 {
                        river_tile[riveramt - 2] = 20;
                    }
                    nodir2 = if dir == 1 { 0 } else { 2 };
                }
                if (dir == 1 && pdir == 3) || (dir == 2 && pdir == 0) {
                    // South←West or East←North
                    if riveramt > 2 {
                        river_tile[riveramt - 2] = 19;
                    }
                    nodir2 = if dir == 1 { 0 } else { 3 };
                }

                pdir = dir;
            } else {
                // Revert position if not floor
                rx = px;
                ry = py;
            }
        }

        // 5. Check if river reached an endpoint
        let mut valid_endpoint = false;
        if dir == 0 && ry >= 2 && dungeon.tiles[rx][ry - 1] == 10 && dungeon.tiles[rx][ry - 2] == 8 {
            river_x[riveramt] = rx;
            river_y[riveramt] = ry - 1;
            river_tile[riveramt] = 24; // North endpoint
            riveramt += 1;
            valid_endpoint = true;
        }
        if dir == 1 && ry < DMAXY - 2 && dungeon.tiles[rx][ry + 1] == 10 && dungeon.tiles[rx][ry + 2] == 8 {
            river_x[riveramt] = rx;
            river_y[riveramt] = ry + 1;
            river_tile[riveramt] = 23; // South endpoint
            riveramt += 1;
            valid_endpoint = true;
        }
        if dir == 2 && rx < DMAXX - 2 && dungeon.tiles[rx + 1][ry] == 10 && dungeon.tiles[rx + 2][ry] == 8 {
            river_x[riveramt] = rx + 1;
            river_y[riveramt] = ry;
            river_tile[riveramt] = if self.flip_coin() { 42 } else { 43 }; // East endpoint
            riveramt += 1;
            valid_endpoint = true;
        }
        if dir == 3 && rx >= 2 && dungeon.tiles[rx - 1][ry] == 10 && dungeon.tiles[rx - 2][ry] == 8 {
            river_x[riveramt] = rx - 1;
            river_y[riveramt] = ry;
            river_tile[riveramt] = if self.flip_coin() { 38 } else { 39 }; // West endpoint
            riveramt += 1;
            valid_endpoint = true;
        }

        // 6. Only place river if valid and long enough
        if valid_endpoint && riveramt >= 7 {
            // Place river tiles
            for k in 0..riveramt {
                dungeon.tiles[river_x[k]][river_y[k]] = river_tile[k];
            }

            // 7. Randomly place a bridge on one river segment
            if riveramt > 3 {
                let bridge_idx = self.random_range(1, riveramt - 1);
                let tile = river_tile[bridge_idx];
                if (tile >= 15 && tile <= 18) || (tile >= 19 && tile <= 22) {
                    // Only place bridges on straight/corner tiles
                    dungeon.tiles[river_x[bridge_idx]][river_y[bridge_idx]] = if tile <= 18 { 44 } else { 45 };
                }
            }

            river_count += 1;
        }
    }
}
```

**关键点**:
- `nodir`/`nodir2` 双重阻塞防止河流折回
- `pdir` 追踪前一方向用于拐角检测
- 8 种拐角模式 (tiles 19-22): NW/NE/SW/SE 拐角
- 端点验证: 必须是 `floor(7) → wall(10) → ceiling(8)` 序列
- 最小长度: ≥7 瓦片才放置河流
- 桥梁随机放置在中间段

### 2. PlacePool 系统 (Spawn/SpawnEdge)

**C++ 等价**: `Spawn()` (lines 1315-1363), `SpawnEdge()` (lines 1282-1313), `PlaceLavaPool()` (lines 1442-1493)

**目的**: 洪水填充检测孤立区域，转换为熔岩池

**SPAWN_TABLE**: 控制洪水填充方向
```rust
const SPAWN_TABLE: [u8; 15] = [
    0x00, 0x0A, 0x03, 0x05, 0x0C, 0x06, 0x09, 0x00, 0x00, 0x0C, 0x03, 0x06, 0x09, 0x0A, 0x05
];
// Bit flags: 0x08=North, 0x04=South, 0x02=East, 0x01=West
```

**POOL_SUB**: 瓦片转换表 (dungeon tile → lava tile)
```rust
const POOL_SUB: [u8; 15] = [
    0, 35, 26, 36, 25, 29, 34, 7, 33, 28, 27, 37, 32, 31, 30
];
```

**实现**:

```rust
fn spawn_edge(&self, dungeon: &mut Dungeon, x: i32, y: i32, totarea: &mut i32) -> bool {
    const SPAWN_TABLE: [u8; 15] = [0x00, 0x0A, 0x03, 0x05, 0x0C, 0x06, 0x09, 0x00, 0x00, 0x0C, 0x03, 0x06, 0x09, 0x0A, 0x05];

    if *totarea > 40 { return true; } // Area too large
    if x < 0 || y < 0 || x >= DMAXX as i32 || y >= DMAXY as i32 { return true; } // Out of bounds
    if (dungeon.tiles[x as usize][y as usize] & 0x80) != 0 { return false; } // Already visited
    if dungeon.tiles[x as usize][y as usize] > 15 { return false; } // Special tile

    let i = dungeon.tiles[x as usize][y as usize];
    dungeon.tiles[x as usize][y as usize] |= 0x80; // Mark visited
    *totarea += 1;

    let st = SPAWN_TABLE[i as usize];
    if (st & 0x80) != 0 && self.spawn_edge(dungeon, x, y - 1, totarea) { return true; }
    if (st & 0x40) != 0 && self.spawn_edge(dungeon, x, y + 1, totarea) { return true; }
    if (st & 0x20) != 0 && self.spawn_edge(dungeon, x + 1, y, totarea) { return true; }
    if (st & 0x10) != 0 && self.spawn_edge(dungeon, x - 1, y, totarea) { return true; }

    false
}

fn spawn(&self, dungeon: &mut Dungeon, x: i32, y: i32, totarea: &mut i32) -> bool {
    const SPAWN_TABLE: [u8; 15] = [0x00, 0x0A, 0x03, 0x05, 0x0C, 0x06, 0x09, 0x00, 0x00, 0x0C, 0x03, 0x06, 0x09, 0x0A, 0x05];

    if *totarea > 40 { return true; }
    if x < 0 || y < 0 || x >= DMAXX as i32 || y >= DMAXY as i32 { return true; }
    if (dungeon.tiles[x as usize][y as usize] & 0x80) != 0 { return false; }
    if dungeon.tiles[x as usize][y as usize] > 15 { return true; }

    let i = dungeon.tiles[x as usize][y as usize];
    dungeon.tiles[x as usize][y as usize] |= 0x80;
    *totarea += 1;

    if i != 8 { // Non-ceiling tiles use SPAWN_TABLE
        let st = SPAWN_TABLE[i as usize];
        if (st & 8) != 0 && self.spawn_edge(dungeon, x, y - 1, totarea) { return true; }
        if (st & 4) != 0 && self.spawn_edge(dungeon, x, y + 1, totarea) { return true; }
        if (st & 2) != 0 && self.spawn_edge(dungeon, x + 1, y, totarea) { return true; }
        if (st & 1) != 0 && self.spawn_edge(dungeon, x - 1, y, totarea) { return true; }
    } else { // Ceiling (tile 8) floods in all directions
        if self.spawn(dungeon, x + 1, y, totarea) { return true; }
        if self.spawn(dungeon, x - 1, y, totarea) { return true; }
        if self.spawn(dungeon, x, y + 1, totarea) { return true; }
        if self.spawn(dungeon, x, y - 1, totarea) { return true; }
    }

    false
}

fn place_lava_pool(&mut self, dungeon: &mut Dungeon) -> bool {
    const POOL_SUB: [u8; 15] = [0, 35, 26, 36, 25, 29, 34, 7, 33, 28, 27, 37, 32, 31, 30];

    let mut lava_pool_placed = false;

    for duny in 0..DMAXY {
        for dunx in 0..DMAXX {
            if dungeon.tiles[dunx][duny] != 8 { continue; }

            dungeon.tiles[dunx][duny] |= 0x80;
            let mut totarea = 1;
            let mut found = true;

            // Check 4 neighbors for connectivity
            if dunx + 1 < DMAXX {
                found = self.spawn(dungeon, (dunx + 1) as i32, duny as i32, &mut totarea);
            }
            if dunx > 0 && !found {
                found = self.spawn(dungeon, (dunx - 1) as i32, duny as i32, &mut totarea);
            } else { found = true; }
            if duny + 1 < DMAXY && !found {
                found = self.spawn(dungeon, dunx as i32, (duny + 1) as i32, &mut totarea);
            } else { found = true; }
            if duny > 0 && !found {
                found = self.spawn(dungeon, dunx as i32, (duny - 1) as i32, &mut totarea);
            } else { found = true; }

            let place_pool = self.random_range(0, 100) < 25; // 25% probability

            // Clear visited flags and optionally convert to lava
            for j in (duny.saturating_sub(totarea as usize))..(duny + totarea as usize).min(DMAXY) {
                for i in (dunx.saturating_sub(totarea as usize))..(dunx + totarea as usize).min(DMAXX) {
                    if (dungeon.tiles[i][j] & 0x80) != 0 {
                        dungeon.tiles[i][j] &= !0x80;
                        if totarea > 4 && place_pool && !found {
                            let k = POOL_SUB[dungeon.tiles[i][j] as usize];
                            if k != 0 && k <= 37 {
                                dungeon.tiles[i][j] = k; // Convert to lava
                            }
                            lava_pool_placed = true;
                        }
                    }
                }
            }
        }
    }

    lava_pool_placed
}
```

**关键点**:
- 使用 `0x80` 标记访问过的瓦片（临时标记）
- `found == false` 表示区域与地图边缘隔离
- `totarea > 4` 确保区域足够大才转换
- `25%` 概率放置熔岩池
- POOL_SUB 转换表处理不同瓦片类型

### 3. CanReplaceTile (特殊瓦片冲突检测)

**C++ 等价**: `CanReplaceTile()` (lines 1367-1391)

**目的**: 防止特殊瓦片 (84-100) 被 miniset 覆盖

**实现**:

```rust
fn can_replace_tile(&self, dungeon: &Dungeon, replace: u8, tx: usize, ty: usize) -> bool {
    if replace < 84 || replace > 100 {
        return true; // Non-special tiles always replaceable
    }

    // Check 3x3 neighborhood for existing special tiles (84-100)
    let check_tile = |x: i32, y: i32| -> bool {
        if x >= 0 && x < DMAXX as i32 && y >= 0 && y < DMAXY as i32 {
            let tile = dungeon.tiles[x as usize][y as usize];
            tile >= 84 && tile <= 100
        } else {
            false
        }
    };

    let tx = tx as i32;
    let ty = ty as i32;

    // Check all 8 neighbors
    if check_tile(tx - 1, ty - 1) || check_tile(tx + 1, ty + 1)
        || check_tile(tx - 1, ty + 1) || check_tile(tx + 1, ty - 1)
        || check_tile(tx, ty - 1) || check_tile(tx, ty + 1)
        || check_tile(tx - 1, ty) || check_tile(tx + 1, ty)
    {
        return false; // Has special tile neighbor
    }

    true
}
```

**C++ Bugfix**: 原版有 `p2` workaround bug，Rust 版本简化为只检查 `p1`

```rust
fn place_miniset_random(&mut self, dungeon: &mut Dungeon, miniset: &Miniset, rndper: usize) -> bool {
    let sw = miniset.width;
    let sh = miniset.height;

    let mut placed = false;
    for sy in 0..(DMAXY - sh) {
        for sx in 0..(DMAXX - sw) {
            if !self.miniset_matches(dungeon, miniset, sx, sy) {
                continue; // Pattern doesn't match
            }
            if !self.can_replace_tile(dungeon, miniset.replace[0][0], sx, sy) {
                continue; // Would conflict with special tile
            }
            if self.random_range(0, 100) >= rndper {
                continue; // Random probability check
            }
            self.place_miniset_at(dungeon, miniset, sx, sy);
            placed = true;
        }
    }

    placed
}

fn place_miniset_random_1x1(&mut self, dungeon: &mut Dungeon, search: u8, replace: u8, rndper: usize) {
    let miniset = Miniset {
        width: 1,
        height: 1,
        search: vec![vec![search]],
        replace: vec![vec![replace]],
    };
    self.place_miniset_random(dungeon, &miniset, rndper);
}
```

**关键点**:
- 遍历整个地图查找匹配位置
- `rndper` 参数控制放置概率 (0-100)
- CanReplaceTile 防止冲突
- 1x1 变体用于单瓦片替换
