pub fn update(&mut self, rng: &mut impl Rng) {
    if self.is_town {

    }

}

### 实现细节

#### Player 处理

```rust
fn process_player(&mut self, rng: &mut impl Rng) {
    // HP 恢复 (每 4 级 +1 HP/tick)
    if self.game_tick % 20 == 0 {
        let regen = (self.player._p_level as i32) / 4 + 1;
        self.player._p_hit_points = (self.player._p_hit_points + regen * 64)
            .min(self.player._p_max_hp);
    }

    // Mana 恢复 (每 3 级 +1 Mana/tick)
    if self.game_tick % 30 == 0 && self.player._p_max_mana > 0 {
        let regen = (self.player._p_level as i32) / 3 + 1;
        self.player._p_mana = (self.player._p_mana + regen * 64)
            .min(self.player._p_max_mana);
    }

    // 检测 Player 攻击怪物
    self.check_player_combat(rng);
}
```

#### Monster 处理

```rust
fn process_monsters(&mut self, rng: &mut impl Rng) {
    // 收集所有活跃怪物 ID (避免 borrow checker 冲突)
    let monster_ids: Vec<usize> = self.monster_manager.iter()
        .map(|(id, _)| id)
        .collect();

    for monster_id in monster_ids {
        // 1. HP 恢复 (每 32 ticks)
        if self.game_tick % 32 == 0 {
            self.regenerate_monster_hp(monster_id);
        }

        // 2. AI 处理 (placeholder - 待 Day 41-45 实现)
        // monster.update_ai(self.player.position, &self.objects, rng);

        // 3. 检查怪物攻击玩家
        self.check_monster_combat(monster_id, rng);

        // 4. 检查怪物开门
        if let Some(monster) = self.monster_manager.get_monster(monster_id) {
            if can_monster_open_doors(monster) {
                monster_check_doors(monster, &mut self.objects);
            }
        }
    }
}
```

### 解决的技术难题

#### 问题 1: MonsterManager API 不匹配

**错误**: 调用不存在的 `get_active_monster_id()` 方法

**原因**: MonsterManager 使用迭代器模式，不提供索引枚举

**解决方案**:
```rust
// ❌ 错误写法
for i in 0..self.monster_manager.active_count() {
    let monster_id = self.monster_manager.get_active_monster_id(i); // 不存在
}

// ✅ 正确写法
let monster_ids: Vec<usize> = self.monster_manager.iter()
    .map(|(id, _)| id)
    .collect();
for monster_id in monster_ids { /* ... */ }
```

#### 问题 2: 类型转换错误

**错误**: `cannot add u8 to i32` (Player._p_level 是 u8)

**解决方案**:
```rust
// ❌ 错误
let regen = self.player._p_level / 4 + 1; // u8 / i32 类型不匹配

// ✅ 正确
let regen = (self.player._p_level as i32) / 4 + 1;
```

#### 问题 3: Borrow Checker 冲突

**错误**: 在遍历 monsters 时无法同时可变借用 player

**解决方案**:
```rust
// ❌ 错误 (同时借用 self.monster_manager 和 self.player)
for (id, monster) in self.monster_manager.iter() {
    monster_attack_player(monster, &mut self.player, rng); // 借用冲突
}

// ✅ 正确 (先收集 ID，再分离处理)
let monster_ids: Vec<usize> = self.monster_manager.iter()
    .map(|(id, _)| id)
    .collect();
for monster_id in monster_ids {
    if let Some(monster) = self.monster_manager.get_monster(monster_id) {
        monster_attack_player(monster, &mut self.player, rng);
    }
}
```

1. **迭代器 vs 索引**: MonsterManager 使用迭代器模式避免无效索引

4. **Town 优化**: 条件分支减少不必要的处理

1. **字段替代**: 当原字段不存在时 (如 `level`)，使用等价字段 (`intelligence`)
