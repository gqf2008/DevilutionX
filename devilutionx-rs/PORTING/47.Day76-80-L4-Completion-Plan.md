```rust
/// Convert dungeon_mask to dungeon tiles using L4_CONV_TABLE
fn make_dmt(&mut self, dungeon: &mut Dungeon) {
    for y in 0..MAXDUNY - 1 {
        for x in 0..MAXDUNX - 1 {
            // 2×2 pattern to index
            let val = ((self.dungeon_mask[x + 1][y + 1] as u8) << 3)
                    | ((self.dungeon_mask[x][y + 1] as u8) << 2)
                    | ((self.dungeon_mask[x + 1][y] as u8) << 1)
                    | (self.dungeon_mask[x][y] as u8);
            dungeon.tiles[x][y] = L4_CONV_TABLE[val as usize];
        }
    }
}
```

```rust
/// Check if horizontal wall can be placed
fn horizontal_wall_ok(&self, dungeon: &Dungeon, i: usize, j: usize) -> Option<usize> {
    let mut x = 1;
    while dungeon.tiles[i + x][j] == 6 {
        if self.protected[i + x][j] { break; }
        if dungeon.tiles[i + x][j - 1] != 6 { break; }
        if dungeon.tiles[i + x][j + 1] != 6 { break; }
        x += 1;
    }

    if matches!(dungeon.tiles[i + x][j], 10 | 12 | 13 | 15 | 16 | 21 | 22) && x > 3 {
        Some(x)
    } else {
        None
    }
}
```

```rust
fn horizontal_wall(&mut self, dungeon: &mut Dungeon, i: usize, j: usize, dx: usize) {
```

```rust
fn add_wall(&mut self, dungeon: &mut Dungeon) {
    for j in 1..MAXDUNY - 1 {
        for i in 1..MAXDUNX - 1 {
            // Try horizontal wall
            if dungeon.tiles[i][j] == 6
                && dungeon.tiles[i - 1][j] == 6

                if let Some(x) = self.horizontal_wall_ok(dungeon, i, j) {

            // Try vertical wall
            if dungeon.tiles[i][j] == 6

                && dungeon.tiles[i][j + 1] == 1 {
                if let Some(y) = self.vertical_wall_ok(dungeon, i, j) {
                    self.vertical_wall(dungeon, i, j, y);
                }
            }
        }
    }
}
```

```rust
/// Calculate total floor area using flood fill
fn find_area(&mut self, dungeon: &Dungeon) -> usize {
    let mut visited = [[false; MAXDUNY]; MAXDUNX];
    let mut area = 0;

    // Find starting floor tile
    let mut start_x = 0usize;
    let mut start_y = 0usize;
    'outer: for y in 0..MAXDUNY {
        for x in 0..MAXDUNX {
            if dungeon.tiles[x][y] == 6 { // Floor
                start_x = x;
                start_y = y;
                break 'outer;
            }
        }
    }

    // Flood fill to count connected floor tiles
    self.flood_area(dungeon, &mut visited, start_x, start_y, &mut area);
    area
}

fn flood_area(&self, dungeon: &Dungeon, visited: &mut [[bool; MAXDUNY]; MAXDUNX],
              x: usize, y: usize, area: &mut usize) {
    if x >= MAXDUNX || y >= MAXDUNY { return; }
    if visited[x][y] { return; }
    if dungeon.tiles[x][y] != 6 { return; } // Not floor

    visited[x][y] = true;
    *area += 1;

    // 4-direction recursion
    if x > 0 { self.flood_area(dungeon, visited, x - 1, y, area); }
    if x < MAXDUNX - 1 { self.flood_area(dungeon, visited, x + 1, y, area); }
    if y > 0 { self.flood_area(dungeon, visited, x, y - 1, area); }
    if y < MAXDUNY - 1 { self.flood_area(dungeon, visited, x, y + 1, area); }
}
```

```rust
pub fn generate(&mut self, dungeon: &mut Dungeon, seed: u32, level: u8, entry: LevelEntry) -> bool {
```
