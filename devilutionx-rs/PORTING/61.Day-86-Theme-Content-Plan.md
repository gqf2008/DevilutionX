```rust
fn theme_barrel(&mut self, dungeon: &mut Dungeon, x: i32, y: i32) {
    // L4 Hell: barrnd[3] = 8 (1/8 chance)
```

```cpp
void Theme_MonstPit(int t) {
    int r = GenerateRnd(100) + 1;
    int ixp = 0, iyp = 0;
    while (r > 0) {
        if (dTransVal[ixp][iyp] == themes[t].ttval && IsTileNotSolid({ ixp, iyp })) {
            --r;
        }
        if (r <= 0) continue;
        ixp++;
        if (ixp == MAXDUNX) {
            ixp = 0;
            iyp++;
            if (iyp == MAXDUNY) iyp = 0;
        }
    }
    CreateRndItem({ ixp, iyp }, true, false, true);
}
```

```rust
fn theme_monster_pit(&mut self, dungeon: &mut Dungeon, x: i32, y: i32) {
    // Find random floor tile in 3x3 room for item
    let mut positions = Vec::with_capacity(9);

    for dy in 0..3 {
        for dx in 0..3 {
```
