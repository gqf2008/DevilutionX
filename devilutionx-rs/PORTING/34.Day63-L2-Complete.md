```rust
fn fill_voids(&mut self) -> bool {
    let mut attempts = 0;

    while self.count_empty_tiles() > 700 && attempts < 100 {
        // 随机墙壁位置
        let xx = random(1..38);
        let yy = random(1..38);

        if self.predungeon[xx][yy] != '#' {
            attempts += 1;
            continue;
        }

        // 检测4种空白模式
        let (xf1, yf1, xf2, yf2) = self.detect_void_pattern(xx, yy);

        if xf1 || yf1 || xf2 || yf2 {
            self.fill_void(xf1, yf1, xf2, yf2, xx, yy);
        }

        attempts += 1;
    }

    self.count_empty_tiles() <= 700
}
```

```rust
// Pattern 1: 左空右满

// Pattern 2: 右空左满
```

## 🎯 Part 2: CreateRoom 递归生成 (Day 63.2)

### 递归房间细分算法

```rust
fn create_room(
    &mut self,
    top_left: Point,
    bottom_right: Point,
    dest_room: usize,      // 目标房间ID
    hall_dir: HallDirection, // Hall方向
    quest_size: Option<(usize, usize)>, // Quest房间尺寸
) {
    const AREA_MIN: i32 = 2;
    const ROOM_MAX: usize = 10;
    const ROOM_MIN: usize = 4;

    // 1. 终止条件检查
    if self.room_count >= 81
        || top_left.x + AREA_MIN > bottom_right.x
        || top_left.y + AREA_MIN > bottom_right.y {
        return;
    }

    // 2. 房间尺寸计算
    let (room_width, room_height) = if let Some((qw, qh)) = quest_size {
        (qw, qh) // Quest房间固定尺寸
    } else {
        // 随机尺寸 (4-10格)
        let w = random_range(ROOM_MIN, area_width.min(ROOM_MAX));
        let h = random_range(ROOM_MIN, area_height.min(ROOM_MAX));
        (w, h)
    };

    // 3. 随机房间位置
    let random_x = random_range(0, area_width);
    let random_y = random_range(0, area_height);
    let room_top_left = top_left + (random_x, random_y);
    let room_bottom_right = room_top_left + (room_width, room_height);

    // 4. 边界调整
    room_top_left = room_top_left.clamp(1, 38);
    room_bottom_right = room_bottom_right.clamp(1, 38);

    // 5. 定义房间 (标记 A/B/C/E 角落)
    self.define_room(room_top_left, room_bottom_right, quest_size.is_some());

    // 6. 添加到房间列表
    let room_id = self.room_count;
    self.room_list.push(RoomNode::new(room_top_left, room_bottom_right));
    self.room_count += 1;

    // 7. 创建Hall连接
    if dest_room != 0 {
        let (hx1, hy1, hx2, hy2) = match hall_dir {
            HallDirection::Up => {
                let hx1 = random_range(0, room_width-2) + room_top_left.x + 1;
                let hy1 = room_top_left.y;
                let dest = &self.room_list[dest_room];
                let hw = (dest.bottom_right.x - dest.top_left.x - 2).max(1);
                let hx2 = random_range(0, hw) + dest.top_left.x + 1;
                let hy2 = dest.bottom_right.y;
                (hx1, hy1, hx2, hy2)
            }
            // ... Down/Left/Right 类似
        };

        self.hall_list.push_back(HallNode::new(
            Point::new(hx1, hy1),
            Point::new(hx2, hy2),
            hall_dir,
        ));
    }

    // 8. 递归细分 (4个子区域)
    const STANDOFF: i32 = 2;

    if room_height > room_width {
        // 垂直切割 (高>宽)
        self.create_room(
            top_left + (STANDOFF, STANDOFF),
            room_bottom_left - (STANDOFF, STANDOFF),
            room_id, HallDirection::Right, None
        );
        self.create_room(
            room_top_right + (STANDOFF, STANDOFF),
            bottom_right - (STANDOFF, STANDOFF),
            room_id, HallDirection::Left, None
        );
        self.create_room(
            (top_left.x, room_bottom_right.y) + (STANDOFF, STANDOFF),
            (room_bottom_right.x, bottom_right.y) - (STANDOFF, STANDOFF),
            room_id, HallDirection::Up, None
        );
        self.create_room(
            (room_top_left.x, top_left.y) + (STANDOFF, STANDOFF),
            (bottom_right.x, room_top_left.y) - (STANDOFF, STANDOFF),
            room_id, HallDirection::Down, None
        );
    } else {
        // 水平切割 (宽>高) - 4个子区域类似
        // ...
    }
}
```

### 辅助方法: define_room()

```rust
fn define_room(&mut self, top_left: Point, bottom_right: Point, is_quest: bool) {
    // 1. 填充房间内部为 '.'
    for y in (top_left.y + 1)..bottom_right.y {
        for x in (top_left.x + 1)..bottom_right.x {
            self.predungeon[x][y] = '.';
        }
    }

    // 2. 标记角落 (非Quest房间)
    if !is_quest {
        self.predungeon[top_left.x][top_left.y] = 'A';           // 左上
        self.predungeon[bottom_right.x][top_left.y] = 'B';       // 右上
        self.predungeon[top_left.x][bottom_right.y] = 'C';       // 左下
        self.predungeon[bottom_right.x][bottom_right.y] = 'E';   // 右下
    }

    // 3. 标记墙壁 '#'
    for x in top_left.x..=bottom_right.x {
        self.predungeon[x][top_left.y] = '#';    // 上墙
        self.predungeon[x][bottom_right.y] = '#'; // 下墙
    }
    for y in top_left.y..=bottom_right.y {
        self.predungeon[top_left.x][y] = '#';       // 左墙
        self.predungeon[bottom_right.x][y] = '#';   // 右墙
    }
}
```

**功能**:
- 房间内部标记为 `.` (地板)
- 4个角落标记为 `A/B/C/E` (用于Hall碰撞检测)
- 房间边界标记为 `#` (墙)

## 🎯 Part 3: DoPatternCheck 模式匹配 (Day 63.3)

### PATTERNS 数据表 (72个模式 + 结束标记)

```rust
const PATTERNS: [[i32; 10]; 73] = [
    // 每个模式: [9个匹配值 (3x3网格), 1个替换值]
    [0, 0, 0, 0, 0, 0, 0, 0, 0, 3],  // Pattern 0
    [0, 0, 0, 0, 2, 0, 0, 0, 0, 3],  // Pattern 1
    [0, 7, 0, 0, 1, 0, 0, 5, 0, 2],  // Pattern 2
    // ... 共72个有效模式
    [1, 1, 1, 1, 1, 1, 1, 1, 2, 8],  // Pattern 71
    [0, 0, 0, 0, 255, 0, 0, 0, 0, 0], // 结束标记
];
```

### 模式匹配值定义

| 值 | 含义 | 匹配条件 |
|----|------|----------|
| 0 | 通配符 | 任意字符都匹配 |
| 1 | 墙 | 必须是 `'#'` |
| 2 | 地板 | 必须是 `'.'` |
| 3 | 门 | 必须是 `'D'` |
| 4 | 空白 | 必须是 `' '` |
| 5 | 门或地板 | `'D'` 或 `'.'` |
| 6 | 门或墙 | `'D'` 或 `'#'` |
| 7 | 空白或地板 | `' '` 或 `'.'` |
| 8 | 门/墙/地板 | `'D'` 或 `'#'` 或 `'.'` |
| 255 | 结束标记 | 位于索引4,标记表结尾 |

### 3x3 网格布局

```
索引布局:
[0] [1] [2]
[3] [4] [5]
[6] [7] [8]

中心位置 [4] 是待检测/替换位置
```

### DoPatternCheck 实现

```rust
fn do_pattern_check(&self, dungeon: &mut Dungeon, i: usize, j: usize) {
    // 遍历所有模式
    for pattern in &PATTERNS {
        if pattern[4] == 255 {
            break; // 结束标记
        }

        // 读取 3x3 邻域
        let mut x = (i as i32) - 1;
        let mut y = (j as i32) - 1;
        let mut matches = true;

        // 检查9个位置
        for l in 0..9 {
            // 每3个元素换行
            if l == 3 || l == 6 {
                y += 1;
                x = (i as i32) - 1;
            }

            // 边界检查
            if x < 0 || x >= DMAXX || y < 0 || y >= DMAXY {
                x += 1;
                continue; // 越界视为匹配
            }

            let cell = self.predungeon[x as usize][y as usize];
            let pattern_val = pattern[l];

            // 模式匹配
            matches &= match pattern_val {
                0 => true,                     // 通配符
                1 => cell == '#',              // 墙
                2 => cell == '.',              // 地板
                3 => cell == 'D',              // 门
                4 => cell == ' ',              // 空白
                5 => cell == 'D' || cell == '.', // 门或地板
                6 => cell == 'D' || cell == '#', // 门或墙
                7 => cell == ' ' || cell == '.', // 空白或地板
                8 => cell == 'D' || cell == '#' || cell == '.', // 任意非空
                _ => false,
            };

            if !matches {
                break; // 提前退出
            }

            x += 1;
        }

        // 匹配成功,替换中心瓦片
        if matches {
            dungeon.tiles[i][j] = pattern[9] as u8;
            return; // 只应用第一个匹配的模式
        }
    }
}
```

**功能**:
- 遍历72个预定义模式
- 读取 predungeon[i][j] 周围 3×3 区域
- 匹配成功后,将 dungeon[i][j] 替换为模式指定值
- 只应用第一个匹配的模式

**C++ 对齐**: 100% ✅

## 🎯 Part 4: FixTilesPatterns 瓦片修正 (Day 63.4)

### 6条邻接修正规则

```rust
fn fix_tiles_patterns(&self, dungeon: &mut Dungeon) {
    for j in 0..(DMAXY - 1) {
        for i in 0..(DMAXX - 1) {
            // Rule 1: Tile 1 + Tile 3 below → 下方改为 1
            if dungeon.tiles[i][j] == 1 && dungeon.tiles[i][j + 1] == 3 {
                dungeon.tiles[i][j + 1] = 1;
            }

            // Rule 2: Tile 3 + Tile 1 below → 下方改为 3
            if dungeon.tiles[i][j] == 3 && dungeon.tiles[i][j + 1] == 1 {
                dungeon.tiles[i][j + 1] = 3;
            }

            // Rule 3: Tile 3 + Tile 7 right → 右侧改为 3
            if dungeon.tiles[i][j] == 3 && dungeon.tiles[i + 1][j] == 7 {
                dungeon.tiles[i + 1][j] = 3;
            }

            // Rule 4: Tile 2 + Tile 3 right → 右侧改为 2
            if dungeon.tiles[i][j] == 2 && dungeon.tiles[i + 1][j] == 3 {
                dungeon.tiles[i + 1][j] = 2;
            }

            // Rule 5: Tile 11 + Tile 14 right → 右侧改为 16
            if dungeon.tiles[i][j] == 11 && dungeon.tiles[i + 1][j] == 14 {
                dungeon.tiles[i + 1][j] = 16;
            }

            // Rule 6: Tile 15 + Tile 1 below → 下方改为 8
            if dungeon.tiles[i][j] == 15 && dungeon.tiles[i][j + 1] == 1 {
                dungeon.tiles[i][j + 1] = 8;
            }
        }
    }
}
```

**功能**: 修正相邻瓦片的视觉不一致问题

## 🎯 Part 5: create_dungeon 集成管道

### 完整生成流程

```rust
fn create_dungeon(&mut self, dungeon: &mut Dungeon) -> bool {
    // 1. Quest房间尺寸
    let quest_room_size = self.get_quest_room_size();

    // 2. 创建初始房间 (递归细分)
    self.create_room(
        Point::new(2, 2),
        Point::new(DMAXX as i32 - 1, DMAXY as i32 - 1),
        0,
        HallDirection::None,
        quest_room_size,
    );

    // 3. 连接所有Hall (BFS)
    while let Some(hall) = self.hall_list.pop_front() {
        self.connect_hall(&hall);
    }

    // 4. 清理 predungeon 特殊字符
    for y in 0..DMAXY {
        for x in 0..DMAXX {
            match self.predungeon[x][y] {
                'A' | 'B' | 'C' | 'E' => self.predungeon[x][y] = '#', // 角落→墙
                ',' => {
                    self.predungeon[x][y] = '.'; // Hall→地板
                    // 标记周围空白为墙
                    for dy in -1..=1 {
                        for dx in -1..=1 {
                            if dx == 0 && dy == 0 { continue; }
                            if predungeon[nx][ny] == ' ' {
                                predungeon[nx][ny] = '#';
                            }
                        }
                    }
                }
                _ => {}
            }
        }
    }

    // 5. FillVoids - 填充空白区域
    if !self.fill_voids() {
        return false; // 失败→重试整个生成
    }

    // 6. DoPatternCheck - 应用模式匹配
    for j in 0..DMAXY {
        for i in 0..DMAXX {
            self.do_pattern_check(dungeon, i, j);
        }
    }

    // 7. FixTilesPatterns - 修正邻接瓦片
    self.fix_tiles_patterns(dungeon);

    true
}
```
