// Price calculation (stores.cpp:1323-1450)
int CalculateSellPrice(const Item &item, int storeGoldMultiplier) {
    int price = item.value;

    // Apply store multiplier (3 for sell, 2 for buy)
    if (storeGoldMultiplier == 1) {
        price = item.value;
    } else {
        price = item.value * storeGoldMultiplier / (storeGoldMultiplier * 2);
    }

    // Durability penalty
    if (item._iMaxDur > 0 && item._iDurability < item._iMaxDur) {
        price = (price * item._iDurability) / item._iMaxDur;
    }

    if (price == 0) price = 1;

    return price;
}

void SetupSmithItems() {
    SmithItems.clear();
    const int items = gbIsHellfire ? NumSmithBasicItemsHf : NumSmithBasicItems;

    for (int i = 0; i < items; i++) {
        SmithItems.emplace_back();
        Item &item = SmithItems.back();

        // Generate weapon/armor based on index
        if (i < 10) {
            // Weapons: Swords, Maces, Axes, Bows
            GenerateItem(item, ItemType::Sword, ...);
        } else {
            // Armor: Shields, Helms, Body armor
            GenerateItem(item, ItemType::Shield, ...);
        }

        item._iCreateInfo = _itemCreateInfo;
        item._iSeed = AdvanceRndSeed();
        item._iIdentified = true;
    }

    // Premium items (6-15 depending on game mode)
    PremiumItemCount = gbIsHellfire ? NumSmithItemsHf : NumSmithItems;
    PremiumItemLevel = MyPlayer->_pLevel;

    for (int i = 0; i < PremiumItemCount; i++) {
        PremiumItems.emplace_back();
        GeneratePremiumItem(PremiumItems[i], PremiumItemLevel);
    }
}

1. **价格计算**
   ```rust
   pub fn calculate_sell_price(
       item: &Item,
       store_multiplier: i32,
   ) -> i32 {
       let mut price = item.value;

       // Apply store multiplier
       // Buy from player: multiplier = 3 (player gets 1/3 value)
       // Sell to player: multiplier = 2 (player pays 2x base value)
       if store_multiplier != 1 {
           price = (price * store_multiplier) / (store_multiplier * 2);
       }

       // Durability penalty
       if item.max_durability > 0 && item.durability < item.max_durability {
           price = (price * item.durability as i32) / item.max_durability as i32;
       }

       if price == 0 {
           price = 1;
       }

       price
   }

   pub fn calculate_buy_price(item: &Item) -> i32 {
       calculate_sell_price(item, 2)  // Player pays 2x
   }

   pub fn calculate_player_sell_price(item: &Item) -> i32 {
       calculate_sell_price(item, 3)  // Player gets 1/3
   }

   pub fn calculate_repair_cost(item: &Item) -> i32 {
       if item.max_durability == 0 {
           return 0;
       }

       let damage = item.max_durability - item.durability;
       let base_price = item.value / item.max_durability as i32;

       base_price * damage as i32
   }
   ```

2. **交易结果枚举**
   ```rust
   pub enum TradeResult {
       Success,
       InsufficientGold,
       InventoryFull,
       ItemNotAvailable,
       InvalidItem,
   }

   pub fn buy_item(
       player: &mut Player,
       shop: &mut ShopInventory,
       item_index: usize,
   ) -> TradeResult {
       let item = match shop.get_item(item_index) {
           Some(item) => item,
           None => return TradeResult::ItemNotAvailable,
       };

       let price = calculate_buy_price(item);

       if player.gold < price {
           return TradeResult::InsufficientGold;
       }

       // TODO: Check inventory space

       player.gold -= price;
       shop.remove_item(item_index);

       TradeResult::Success
   }

   pub fn sell_item(
       player: &mut Player,
       shop: &mut ShopInventory,
       item: Item,
   ) -> TradeResult {
       let price = calculate_player_sell_price(&item);

       if shop.add_item(item).is_err() {
           return TradeResult::InventoryFull;
       }

       player.gold += price;

       TradeResult::Success
   }
   ```
