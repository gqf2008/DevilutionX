// In town.rs (Day 90将集成):
pub fn create_town_with_quests(
    entry: TownEntry,
    twarp_from: i32,
    quests: &[Quest; MAX_QUESTS]
) -> TownGenerator {
    let mut gen = create_town(entry, twarp_from);

    // Hive Quest (Farmer)
    let farmer_quest = &quests[QuestId::Farmer as usize];
    match farmer_quest.state {
        QuestState::Init | QuestState::HiveTease1 | QuestState::HiveTease2 => {
            gen.town_close_hive();
        },
        QuestState::Active | QuestState::HiveActive => {
            gen.town_open_hive();
        },
        _ => {}
    }

    gen
}
