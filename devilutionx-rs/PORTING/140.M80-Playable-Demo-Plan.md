```rust
pub struct GameState {
    running: bool,
    paused: bool,
    delta_time: f64,
    frame_count: u64,
    player_pos: Point,
    camera_pos: Point,
}

pub struct GameLoop {
    window: GameWindow,
    state: GameState,
    input: InputSystem,
    fps_target: u32,
}

impl GameLoop {
    pub fn new() -> Result<Self>;

    pub fn run(&mut self) -> Result<()> {
        while self.state.running {
            let frame_start = Instant::now();

            self.process_input()?;
            self.update(self.state.delta_time)?;
            self.render()?;

            self.wait_for_frame(frame_start);
        }
        Ok(())
    }

    fn process_input(&mut self) -> Result<()>;
    fn update(&mut self, dt: f64) -> Result<()>;
    fn render(&mut self) -> Result<()>;
}
```

```rust
pub struct MapRenderer {
    tileset: TileSet,
    tiles: Vec<Vec<u8>>,  // 瓦片索引
    width: usize,
    height: usize,
}

impl MapRenderer {
    pub fn load_from_dun(path: &Path, resources: &ResourceManager) -> Result<Self>;

    /// 渲染可见瓦片 (C++ DrawFloor/DrawTileContent对应)
    pub fn render(&self, surface: &mut Surface, camera: Point, viewport: Size) {
        // 计算可见瓦片范围
        let start_tile = self.screen_to_tile(camera);
        let visible_cols = (viewport.width / TILE_WIDTH) + 2;
        let visible_rows = (viewport.height / TILE_HEIGHT) + 4;

        // 等距渲染 (菱形遍历)
        for row in 0..visible_rows {
            for col in 0..visible_cols {
                let tile_x = start_tile.x + col - row;
                let tile_y = start_tile.y + row;

                if let Some(tile_id) = self.get_tile(tile_x, tile_y) {
                    let screen_pos = self.tile_to_screen(tile_x, tile_y, camera);
                    self.draw_tile(surface, tile_id, screen_pos);
                }
            }
        }
    }

    fn tile_to_screen(&self, tile_x: i32, tile_y: i32, camera: Point) -> Point {
        // 等距坐标转换
        let iso_x = (tile_x - tile_y) * (TILE_WIDTH / 2);
        let iso_y = (tile_x + tile_y) * (TILE_HEIGHT / 2);
        Point { x: iso_x - camera.x, y: iso_y - camera.y }
    }

    fn screen_to_tile(&self, screen_pos: Point) -> Point {
        // 逆向等距转换
        let tile_x = (screen_pos.x / (TILE_WIDTH / 2) + screen_pos.y / (TILE_HEIGHT / 2)) / 2;
        let tile_y = (screen_pos.y / (TILE_HEIGHT / 2) - screen_pos.x / (TILE_WIDTH / 2)) / 2;
        Point { x: tile_x, y: tile_y }
    }
}
```

## 技术细节

### 等距坐标系

```
世界坐标 (tile_x, tile_y) → 屏幕坐标 (screen_x, screen_y)

screen_x = (tile_x - tile_y) * 32
screen_y = (tile_x + tile_y) * 16

逆向:
tile_x = (screen_x / 32 + screen_y / 16) / 2
tile_y = (screen_y / 16 - screen_x / 32) / 2
```

### 帧率控制

```rust
const TARGET_FPS: u32 = 60;
const FRAME_TIME: Duration = Duration::from_micros(16_666); // 1/60秒

fn wait_for_frame(&self, frame_start: Instant) {
    let elapsed = frame_start.elapsed();
    if elapsed < FRAME_TIME {
        std::thread::sleep(FRAME_TIME - elapsed);
    }
}
```

### 精灵绘制 (带透明)

```rust
fn draw_sprite_cl2(surface: &mut Surface, sprite: &Surface, pos: Point, palette: &Palette) {
    for y in 0..sprite.height {
        for x in 0..sprite.width {
            let color = sprite.get_pixel(x, y);
            if color.a > 0 {  // 跳过透明像素
                surface.set_pixel(pos.x + x, pos.y + y, color);
            }
        }
    }
}
```
