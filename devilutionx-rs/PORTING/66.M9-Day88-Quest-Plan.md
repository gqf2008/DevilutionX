impl QuestManager {
    /// Check and trigger quests based on game events
    pub fn check_quest_trigger(&mut self, event: QuestEvent) {
        // C++ CheckQuests() in quests.cpp:580-644
        match event {
            QuestEvent::EnterLevel(level) => {
                // Check level-based quest triggers
                for quest in &mut self.quests {
                    if quest.level == level && quest.state == QuestState::Init {
                        quest.state = QuestState::Active;
                    }
                }
            },
            QuestEvent::TalkToNPC(npc_id) => {
                // Trigger quests from NPC conversations

                    NpcId::Farnham => {
                        // Butcher quest trigger

                            self.quests[QuestId::Butcher as usize].state = QuestState::Active;

            },
            QuestEvent::KillMonster(monster_id) => {
                // Quest completion triggers

                    },
                    MonsterId::SkelKing => {
                        self.complete_quest(QuestId::SkelKing);
                    },
                    // ...
                }
            },
            QuestEvent::PickupItem(item_id) => {
                // Item pickup quest updates
            },
        }
    }

    /// Complete a quest
    fn complete_quest(&mut self, id: QuestId) {
        let quest = &mut self.quests[id as usize];
        if quest.state == QuestState::Active {
            quest.state = QuestState::Done;
            quest.show_in_log = true;
        }
    }

    /// Update Town map based on quest states (integrate with town.rs)
    pub fn update_town_map(&self, town_gen: &mut TownGenerator) {
        // C++ quests.cpp integration with town.cpp

        // Hive Quest (Hellfire)
        let farmer_quest = &self.quests[QuestId::Farmer as usize];
        match farmer_quest.state {
            QuestState::Init | QuestState::HiveTease1 | QuestState::HiveTease2 => {
                town_gen.town_close_hive();
            },
            QuestState::Active | QuestState::HiveActive => {
                town_gen.town_open_hive();
            },
            _ => {}
        }

        // Grave Quest (Hellfire)
        let grave_quest = &self.quests[QuestId::Grave as usize];
        if grave_quest.state == QuestState::Active {
            town_gen.town_open_grave();
        } else {
            town_gen.town_close_grave();
        }
    }

impl QuestManager {
    /// Get list of quests to display in quest log
    pub fn get_log_quests(&self) -> Vec<(QuestId, &Quest)> {
        let mut quests: Vec<_> = self.quests.iter()
            .enumerate()
            .filter(|(_, q)| q.show_in_log && q.state != QuestState::NotAvail)
            .map(|(i, q)| (QuestId::from_i8(i as i8).unwrap(), q))
            .collect();

        // Sort by book order
        quests.sort_by_key(|(id, _)| QUEST_DATA[*id as usize].book_order);

        quests
    }

    /// Check if quest log should be open
    pub fn is_log_open(&self) -> bool {
        self.quest_log_open
    }

    /// Toggle quest log
    pub fn toggle_log(&mut self) {
        self.quest_log_open = !self.quest_log_open;
    }

    /// Get quest description text ID
    pub fn get_quest_text(&self, id: QuestId) -> u16 {
        QUEST_DATA[id as usize].text_id
    }
}
